%% PUNTO 7: Transformación interactiva de señales
 
clear
clc
close all
 
%% Señal x(t)
 
xf = @(t) ...
1.*(t>=-5 & t<-3) + ...
2.*(t>=-3 & t<-1) + ...
1.5.*(t>=-1 & t<2) + ...
0.5.*(t>=2 & t<4);
 
%% Vector de tiempo
 
t = -22:0.001:22;
 
%% Ventana principal
 
fig = uifigure( ...
'Name','Transformación de señales x(t)', ...
'Position',[100 100 1100 650]);
 
%% Ejes de la gráfica
 
ax = uiaxes(fig, ...
'Position',[300 100 760 500]);
 
grid(ax,'on')
 
xlabel(ax,'Tiempo [s]')
ylabel(ax,'Amplitud')
 
title(ax,'Transformación interactiva de x(t)')
 
xlim(ax,[-22 22])
ylim(ax,[-0.5 2.5])
 
xticks(ax,-22:2:22)
yticks(ax,-0.5:0.5:2.5)
 
%% Panel de controles
 
panel = uipanel(fig, ...
'Title','Controles', ...
'Position',[20 100 250 500]);
 
%% =========================================================
% TRASLACIÓN
%% =========================================================
 
checkTras = uicheckbox(panel, ...
'Text','Activar traslación', ...
'Position',[20 440 180 25], ...
'Value',false);
 
uilabel(panel, ...
'Text','Desplazamiento k:', ...
'Position',[20 395 150 22]);
 
campoK = uieditfield(panel,'numeric', ...
'Position',[20 365 100 25], ...
'Value',0, ...
'Limits',[-5 5]);
 
uilabel(panel, ...
'Text','Slider de k:', ...
'Position',[20 330 150 22]);
 
sliderK = uislider(panel, ...
'Position',[20 315 200 3], ...
'Limits',[-5 5], ...
'Value',0, ...
'MajorTicks',-5:1:5);
 
%% =========================================================
% REFLEXIÓN
%% =========================================================
 
checkRef = uicheckbox(panel, ...
'Text','Activar reflexión', ...
'Position',[20 260 180 25], ...
'Value',false);
 
%% =========================================================
% ESCALAMIENTO
%% =========================================================
 
checkEsc = uicheckbox(panel, ...
'Text','Activar escalamiento', ...
'Position',[20 215 180 25], ...
'Value',false);
 
uilabel(panel, ...
'Text','Factor a:', ...
'Position',[20 180 100 22]);
 
campoA = uieditfield(panel,'numeric', ...
'Position',[20 150 100 25], ...
'Value',1, ...
'Limits',[1/3 3]);
 
uilabel(panel, ...
'Text','Slider de a:', ...
'Position',[20 115 150 22]);
 
sliderA = uislider(panel, ...
'Position',[20 100 200 3], ...
'Limits',[1/3 3], ...
'Value',1, ...
'MajorTicks',[1/3 0.5 1 2 3]);
 
%% =========================================================
% TEXTO DE LA ECUACIÓN
%% =========================================================
 
textoOperacion = uilabel(fig, ...
'Position',[320 30 700 40], ...
'Text','y(t)=x(t)', ...
'HorizontalAlignment','center', ...
'FontSize',14);
 
%% =========================================================
% CALLBACKS
%% =========================================================
 
callbackFcn = @(src,event) actualizarGrafica( ...
src,event,ax,xf,t,...
checkTras,checkRef,checkEsc,...
sliderK,sliderA,campoK,campoA,...
textoOperacion);
 
checkTras.ValueChangedFcn = callbackFcn;
checkRef.ValueChangedFcn = callbackFcn;
checkEsc.ValueChangedFcn = callbackFcn;
 
sliderK.ValueChangedFcn = callbackFcn;
sliderA.ValueChangedFcn = callbackFcn;
 
campoK.ValueChangedFcn = callbackFcn;
campoA.ValueChangedFcn = callbackFcn;
 
%% =========================================================
% GRÁFICA INICIAL
%% =========================================================
 
actualizarGrafica([],[],...
ax,xf,t,...
checkTras,checkRef,checkEsc,...
sliderK,sliderA,campoK,campoA,...
textoOperacion);
 
%% =========================================================
% FUNCIÓN DE ACTUALIZACIÓN
%% =========================================================
 
function actualizarGrafica(src,~,ax,xf,t,...
checkTras,checkRef,checkEsc,...
sliderK,sliderA,campoK,campoA,...
textoOperacion)
 
%% =====================================================
% TRASLACIÓN
%% =====================================================
 
if checkTras.Value
 
% Si se modificó el campo numérico
if isequal(src,campoK)
 
k = campoK.Value;
sliderK.Value = k;
 
% Si se modificó el slider
elseif isequal(src,sliderK)
 
k = sliderK.Value;
campoK.Value = k;
 
% Inicialización
else
 
k = sliderK.Value;
campoK.Value = k;
 
end
 
else
 
k = 0;
 
end
 
%% =====================================================
% ESCALAMIENTO
%% =====================================================
 
if checkEsc.Value
 
% Si se modificó el campo numérico
if isequal(src,campoA)
 
a = campoA.Value;
sliderA.Value = a;
 
% Si se modificó el slider
elseif isequal(src,sliderA)
 
a = sliderA.Value;
campoA.Value = a;
 
% Inicialización
else
 
a = sliderA.Value;
campoA.Value = a;
 
end
 
else
 
a = 1;
 
end
 
%% =====================================================
% REFLEXIÓN
%% =====================================================
 
if checkRef.Value
 
signo = -1;
 
else
 
signo = 1;
 
end
 
%% =====================================================
% SEÑAL TRANSFORMADA
%% =====================================================
 
argumento = signo*a*(t-k);
 
y = xf(argumento);
 
%% =====================================================
% GRÁFICA
%% =====================================================
 
cla(ax)
 
plot(ax,t,y,'b','LineWidth',2)
 
hold(ax,'on')
 
plot(ax,t,xf(t),'r--','LineWidth',1.5)
 
hold(ax,'off')
 
legend(ax,{'Transformada','Original'}, ...
'Location','northoutside')
 
grid(ax,'on')
 
xlabel(ax,'Tiempo [s]')
ylabel(ax,'Amplitud')
 
title(ax,'Transformación interactiva de x(t)')
 
xlim(ax,[-22 22])
ylim(ax,[-0.5 2.5])
 
xticks(ax,-22:2:22)
yticks(ax,-0.5:0.5:2.5)
 
%% =====================================================
% CONSTRUCCIÓN DE LA ECUACIÓN
%% =====================================================
 
% Traslación
 
if checkTras.Value
 
if k >= 0
 
textoTras = sprintf('(t-%g)',k);
 
else
 
textoTras = sprintf('(t+%g)',abs(k));
 
end
 
else
 
textoTras = 't';
 
end
 
% Escalamiento
 
if checkEsc.Value
 
if abs(a-1) < 0.001
 
textoEsc = textoTras;
 
else
 
textoEsc = sprintf('%g%s',a,textoTras);
 
end
 
else
 
textoEsc = textoTras;
 
end
 
% Reflexión
 
if checkRef.Value
 
argumentoTexto = ['-(' textoEsc ')'];
 
else
 
argumentoTexto = textoEsc;
 
end
 
%% Mostrar ecuación
 
textoOperacion.Text = ...
['y(t)=x(' argumentoTexto ')'];
 
end
